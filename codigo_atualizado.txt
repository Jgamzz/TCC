def _load_local_env() -> None:
    """Load local development settings without overriding non-empty environment values."""
    env_file = Path(__file__).resolve().parent.parent / ".env"
    if not env_file.is_file():
        return

    for line in env_file.read_text(encoding="utf-8").splitlines():
        line = line.strip()
        if not line or line.startswith("#") or "=" not in line:
            continue
        key, value = (part.strip() for part in line.split("=", 1))
        if value[:1] in {"'", '"'} and value[-1:] == value[:1]:
            value = value[1:-1]
        if key and not os.environ.get(key):
            os.environ[key] = value


_load_local_env()

import logging
import os
import traceback
from logging.handlers import RotatingFileHandler

from flask import has_request_context, request

from app import db
from app.models.log_sistema import LogSistema


class DatabaseLogHandler(logging.Handler):
    def emit(self, record):
        try:
            db.session.rollback()
            usuario_id = None
            rota = None
            metodo = None
            if has_request_context():
                rota = request.path
                metodo = request.method
                usuario_id = getattr(getattr(request, "usuario", None), "id", None)

            log = LogSistema(
                nivel=record.levelname,
                mensagem=record.getMessage(),
                modulo=record.module,
                metodo=metodo,
                rota=rota,
                usuario_id=usuario_id,
                stack_trace=(
                    "".join(traceback.format_exception(*record.exc_info))
                    if record.exc_info
                    else None
                ),
            )
            db.session.add(log)
            db.session.commit()
        except Exception:
            db.session.rollback()
            self.handleError(record)


def configurar_logging(app):
    os.makedirs("logs", exist_ok=True)
    arquivo = os.path.join("logs", "app.log")
    formato = logging.Formatter(
        "%(asctime)s | %(levelname)s | %(name)s | %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S",
    )
    arquivo_handler = RotatingFileHandler(
        arquivo, maxBytes=5 * 1024 * 1024, backupCount=3, encoding="utf-8"
    )
    arquivo_handler.setFormatter(formato)
    arquivo_handler.setLevel(logging.INFO)

    banco_handler = DatabaseLogHandler()
    banco_handler.setLevel(logging.ERROR)

    app.logger.setLevel(logging.INFO)
    app.logger.addHandler(arquivo_handler)
    app.logger.addHandler(banco_handler)
    logging.getLogger("werkzeug").addHandler(arquivo_handler)


def registrar_erro(app, erro):
    app.logger.error(
        "Erro nÃ£o tratado na API",
        exc_info=(type(erro), erro, erro.__traceback__),
    )

class AuthController:
    """Endpoints pÃºblicos relacionados Ã  autenticaÃ§Ã£o."""

    blueprint = Blueprint("auth_bp", __name__)

    @staticmethod
    def _serializer():
        return URLSafeTimedSerializer(
            current_app.config["JWT_SECRET_KEY"],
            salt=current_app.config["PASSWORD_RESET_SALT"],
        )

    @staticmethod
    @blueprint.post("/login")
    def login():
        """
        Autentica um usuÃ¡rio e retorna um JWT.
        ---
        tags:
          - AutenticaÃ§Ã£o
        consumes:
          - application/json
        parameters:
          - in: body
            name: credenciais
            required: true
            schema:
              type: object
              required: [Username, Password]
              properties:
                Username:
                  type: string
                  example: admin
                Password:
                  type: string
                  format: password
                  example: senha123
        responses:
          200:
            description: Token JWT gerado
          401:
            description: Credenciais invÃ¡lidas
        """
        try:
            login_dto = LoginRequestDto.from_dict(request.get_json(silent=True))
        except ValueError as error:
            return jsonify({"erro": str(error)}), 400

        usuario, _erro_credencial = UsuarioService.autenticar(
            login_dto.username, login_dto.password
        )
        if not usuario:
            return (
                jsonify(
                    {
                        "erro": "UsuÃ¡rio e senha incorretos.",
                        "campos": ["usuario", "senha"],
                    }
                ),
                401,
            )
        if usuario.cargo_id is None:
            return (
                jsonify({"erro": "UsuÃ¡rio autenticado, mas sem cargo definido."}),
                403,
            )

        token = create_access_token(
            identity=str(usuario.id),
            additional_claims={"cargo_id": usuario.cargo_id},
        )
        return jsonify(
            TokenResponseDto(access_token=token, usuario=usuario.to_dict()).to_dict()
        )

    @staticmethod
    @blueprint.post("/recuperar-senha")
    def solicitar_recuperacao():
        dados = request.get_json(silent=True)
        if not isinstance(dados, dict):
            return jsonify({"erro": "Informe um objeto JSON."}), 400
        email = str(dados.get("email", "")).strip().lower()
        if not email:
            return jsonify({"erro": "E-mail Ã© obrigatÃ³rio."}), 400
        resposta = {
            "mensagem": "Se o e-mail estiver cadastrado, enviaremos as instruÃ§Ãµes."
        }
        smtp_host = current_app.config.get("SMTP_HOST")
        remetente = current_app.config.get("SMTP_FROM")
        if not smtp_host or not remetente:
            current_app.logger.error("RecuperaÃ§Ã£o de senha sem SMTP configurado.")
            return jsonify({"erro": "ServiÃ§o de recuperaÃ§Ã£o indisponÃ­vel."}), 503
        from app import db
        from app.models.usuario import Usuario

        usuario = Usuario.query.filter_by(email=email, is_active=True).first()
        if not usuario:
            return jsonify(resposta), 202
        agora = datetime.utcnow()
        if (
            usuario.password_reset_requested_at
            and usuario.password_reset_requested_at > agora - timedelta(minutes=1)
        ):
            return jsonify(resposta), 202
        token = AuthController._serializer().dumps(email)
        link = f"{current_app.config['APP_PUBLIC_URL'].rstrip('/')}/?reset_token={token}"
        mensagem = EmailMessage()
        mensagem["Subject"] = "RedefiniÃ§Ã£o de senha do SGB Varejo"
        mensagem["From"] = remetente
        mensagem["To"] = email
        mensagem.set_content(
            "Foi solicitada a redefiniÃ§Ã£o da senha. "
            f"Acesse este link em atÃ© 15 minutos: {link}"
        )
        with smtplib.SMTP(
            smtp_host, current_app.config["SMTP_PORT"], timeout=10
        ) as servidor:
            servidor.starttls()
            username = current_app.config.get("SMTP_USERNAME")
            password = current_app.config.get("SMTP_PASSWORD")
            if username and password:
                servidor.login(username, password)
            servidor.send_message(mensagem)
        usuario.password_reset_requested_at = agora
        db.session.commit()
        return jsonify(resposta), 202

    @staticmethod
    @blueprint.post("/redefinir-senha")
    def redefinir_senha():
        dados = request.get_json(silent=True)
        if not isinstance(dados, dict):
            return jsonify({"erro": "Informe um objeto JSON."}), 400
        token = dados.get("token")
        nova_senha = dados.get("nova_senha")
        if not isinstance(token, str) or not isinstance(nova_senha, str):
            return jsonify({"erro": "Token e nova senha sÃ£o obrigatÃ³rios."}), 400

            email = AuthController._serializer().loads(

                max_age=current_app.config["PASSWORD_RESET_MAX_AGE_SECONDS"],
            )
        except (BadSignature, SignatureExpired):
            return jsonify({"erro": "O link expirou ou Ã© invÃ¡lido."}), 400
        try:
            UsuarioService.redefinir_senha(email, nova_senha)
        except ValueError as erro:
            return jsonify({"erro": str(erro)}), 400
        return jsonify({"mensagem": "Senha redefinida com sucesso."})

@operacao_bp.get("/relatorios")
@verificar_permissao(["GERENTE", "ADMIN"])
def gerar_relatorio():
    tipo = request.args.get("tipo", "").strip().lower()
    try:
        inicio = datetime.fromisoformat(request.args["inicio"]).date()
        fim = datetime.fromisoformat(request.args["fim"]).date()
    except (KeyError, TypeError, ValueError):
        return jsonify({"erro": "Informe inÃ­cio e fim no formato AAAA-MM-DD."}), 400
    if inicio > fim:
        return jsonify({"erro": "A data inicial nÃ£o pode ser posterior Ã  final."}), 400
    if (fim - inicio).days > 3660:
        return jsonify({"erro": "O perÃ­odo mÃ¡ximo permitido Ã© de 10 anos."}), 400
    inicio_dt = datetime.combine(inicio, datetime.min.time())
    fim_dt = datetime.combine(fim + timedelta(days=1), datetime.min.time())

    if tipo == "vendas":
        linhas = (
            db.session.query(
                Produto.id_produto,
                Produto.nome,
                func.coalesce(func.sum(Venda.quantidade), 0),
                func.coalesce(
                    func.sum(Venda.quantidade * Venda.preco_unitario), 0
                ),
            )
            .outerjoin(
                Venda,
                (Venda.id_produto == Produto.id_produto)
                & (Venda.data_venda >= inicio_dt)
                & (Venda.data_venda < fim_dt),
            )
            .filter(Produto.ativo.is_(True))
            .group_by(Produto.id_produto, Produto.nome)
            .order_by(Produto.nome)
            .all()
        )
        return jsonify(
            {
                "tipo": tipo,
                "inicio": inicio.isoformat(),
                "fim": fim.isoformat(),
                "dados": [
                    {
                        "id_produto": row[0],
                        "produto": row[1],
                        "quantidade": float(row[2]),
                        "faturamento": float(row[3]),
                    }
                    for row in linhas
                ],
            }
        )
    if tipo == "giro":
        linhas = (
            db.session.query(
                Produto.id_produto,
                Produto.nome,
                Produto.estoque,
                func.coalesce(func.sum(Venda.quantidade), 0),
            )
            .outerjoin(
                Venda,
                (Venda.id_produto == Produto.id_produto)
                & (Venda.data_venda >= inicio_dt)
                & (Venda.data_venda < fim_dt),

            .filter(Produto.ativo.is_(True))

                    for row in linhas
                ],

    if tipo == "precos":
        alteracoes = (
            db.session.query(HistoricoPreco, Produto.nome)
            .join(Produto, Produto.id_produto == HistoricoPreco.id_produto)

                        "produto": nome,

from dataclasses import dataclass
from datetime import date
from decimal import Decimal
from typing import Any, Dict, Optional


@dataclass(frozen=True)
class ProdutoRequestDto:
    nome: str
    custo: Decimal
    preco_atual: Decimal
    estoque: Decimal = Decimal("0")
    data_validade: Optional[date] = None
    codigo: Optional[str] = None
    descricao: Optional[str] = None
    categoria: Optional[str] = None
    fornecedor: Optional[str] = None

    @classmethod
    def from_dict(cls, data: Optional[Dict[str, Any]]) -> "ProdutoRequestDto":
        if not isinstance(data, dict):
            raise ValueError("Os dados do produto devem ser um objeto JSON.")
        campos = ("nome", "custo", "preco_atual")
        ausentes = [campo for campo in campos if campo not in data]
        if ausentes:
            raise ValueError("Campos obrigatÃ³rios ausentes: " + ", ".join(ausentes))
        try:
            produto = cls(
                nome=str(data["nome"]).strip(),
                custo=Decimal(str(data["custo"])),
                preco_atual=Decimal(str(data["preco_atual"])),
                estoque=Decimal(str(data.get("estoque", 0))),
                data_validade=cls._parse_date(data.get("data_validade")),
                codigo=(cls._optional_text(data.get("codigo")) or "").upper() or None,
                descricao=cls._optional_text(data.get("descricao")),
                categoria=cls._optional_text(data.get("categoria")),
                fornecedor=cls._optional_text(data.get("fornecedor")),
            )
        except (ArithmeticError, TypeError, ValueError) as error:
            raise ValueError("Os valores numÃ©ricos do produto sÃ£o invÃ¡lidos.") from error
        if not produto.nome:
            raise ValueError("O nome do produto Ã© obrigatÃ³rio.")
        if len(produto.nome) > 150:
            raise ValueError("O nome do produto deve ter atÃ© 150 caracteres.")
        if produto.codigo and len(produto.codigo) > 80:
            raise ValueError("O cÃ³digo deve ter atÃ© 80 caracteres.")
        if produto.categoria and len(produto.categoria) > 100:
            raise ValueError("A categoria deve ter atÃ© 100 caracteres.")
        if produto.fornecedor and len(produto.fornecedor) > 150:
            raise ValueError("O fornecedor deve ter atÃ© 150 caracteres.")
        if (
            not produto.custo.is_finite()
            or not produto.preco_atual.is_finite()
            or not produto.estoque.is_finite()
            or produto.custo < 0
            or produto.preco_atual <= 0
            or produto.estoque < 0
        ):
            raise ValueError("Custo e estoque nÃ£o podem ser negativos; preÃ§o deve ser positivo.")
        return produto

    @staticmethod
    def _optional_text(value):
        return str(value).strip() or None if value is not None else None

    @staticmethod
    def _parse_date(value: Any) -> Optional[date]:
        if value in (None, ""):
            return None
        if isinstance(value, date):
            return value
        try:
            return date.fromisoformat(value)
        except (TypeError, ValueError) as error:
            raise ValueError(
                "data_validade deve estar no formato AAAA-MM-DD."
            ) from error


@dataclass(frozen=True)
class ProdutoResponseDto:
    id_produto: int

    estoque: float
    data_validade: Optional[str]
    codigo: Optional[str]

def _create_database_if_needed(app: Any) -> None:
    """Cria o database MySQL antes de o SQLAlchemy abrir a conexÃ£o principal."""
    uri = app.config["SQLALCHEMY_DATABASE_URI"]
    if not str(uri).startswith("mysql"):
        return

    if not app.config.get("DB_PASSWORD"):
        raise RuntimeError(
            "DB_PASSWORD nÃ£o estÃ¡ definida. Configure a senha do MySQL "
            "no ambiente antes de iniciar a aplicaÃ§Ã£o."
        )

    server_url = URL.create(
        drivername="mysql+pymysql",
        username=app.config["DB_USER"],
        password=app.config["DB_PASSWORD"],
        host=app.config["DB_HOST"],
        port=int(app.config["DB_PORT"]),
        query={"charset": "utf8mb4"},
    )
    engine = create_engine(server_url)
    try:
        with engine.begin() as connection:
            database_name = app.config["DB_NAME"].replace("`", "``")
            connection.execute(
                text(
                    f"CREATE DATABASE IF NOT EXISTS `{database_name}` "
                    "CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci"
                )
            )
    finally:
        engine.dispose()


def _include_object(
    object_: Any,
    name: str,
    object_type: str,
    reflected: bool,
    compare_to: Any,
) -> bool:
    """NÃ£o remove objetos existentes que nÃ£o estejam descritos pelo ORM."""
    if object_type == "table" and reflected and compare_to is None:
        return False
    if object_type == "column" and reflected and compare_to is None:
        return False
    return True


def _ensure_user_auth_columns(connection: Any) -> None:
    """Adiciona colunas de autenticaÃ§Ã£o ausentes em tabelas Usuario legadas."""
    tables = {name.lower(): name for name in inspect(connection).get_table_names()}
    table_name = tables.get(Usuario.__tablename__.lower())
    if table_name is None:
        return

    existing_columns = {
        column["name"].lower()
        for column in inspect(connection).get_columns(table_name)
    }
    additions = (
        Column("Email", String(254), nullable=True),
        Column(
            "failed_login_attempts",
            SmallInteger,
            nullable=False,
            server_default="0",
        ),
        Column("locked_until", DateTime, nullable=True),
        Column("password_reset_requested_at", DateTime, nullable=True),
    )
    operations = Operations(MigrationContext.configure(connection))
    for column in additions:
        if column.name.lower() not in existing_columns:
            operations.add_column(table_name, column)


def _ensure_stock_user_column(connection: Any) -> None:
    """Adiciona o responsÃ¡vel por movimentaÃ§Ãµes em tabelas Estoque legadas."""
    tables = {name.lower(): name for name in inspect(connection).get_table_names()}
    table_name = tables.get("estoque")
    if table_name is None:
        return

    existing_columns = {
        column["name"].lower()
        for column in inspect(connection).get_columns(table_name)
    }
    if "usuario_id" not in existing_columns:
        operations = Operations(MigrationContext.configure(connection))
        operations.add_column(
            table_name,
            Column("usuario_id", Integer, nullable=True),
        )


def _apply_operations(
    operations: Operations,
    operation_list: list[Any],
    existing_tables: set[str],
    existing_indexes: dict[str, set[str]],
) -> None:
    """Aplica operaÃ§Ãµes geradas, incluindo as agrupadas por tabela."""
    for operation in operation_list:
        nested_operations = getattr(operation, "ops", None)
        if nested_operations is not None:
            _apply_operations(
                operations, nested_operations, existing_tables, existing_indexes
            )
        elif (
            operation.__class__.__name__ == "CreateTableOp"
            and operation.table_name.lower() in existing_tables
        ):
            # MySQL costuma retornar nomes em minÃºsculas, embora o ORM preserve
            # a capitalizaÃ§Ã£o de tabelas legadas como "Cargo" e "Usuario".
            continue
        elif (
            operation.__class__.__name__ == "CreateIndexOp"
            and operation.index_name.lower()
            in existing_indexes.get(operation.table_name.lower(), set())
        ):
            continue
        else:
            operations.invoke(operation)


def _get_existing_indexes(connection: Any) -> dict[str, set[str]]:
    """ObtÃ©m Ã­ndices existentes para evitar criaÃ§Ãµes duplicadas no MySQL."""
    inspector = inspect(connection)
    indexes: dict[str, set[str]] = {}
    for table_name in inspector.get_table_names():
        indexes[table_name.lower()] = {
            index["name"].lower()
            for index in inspector.get_indexes(table_name)
            if index.get("name")
        }
    return indexes


def _rename_roles(connection: Any) -> None:
    """Atualiza os nomes legados dos cargos sem alterar seus IDs."""
    connection.execute(
        text(
            "UPDATE Cargo SET Name = :new_name "
            "WHERE UPPER(Name) = :old_name"
        ),
        {"new_name": "Admin", "old_name": "DONO"},
    )
    connection.execute(
        text(
            "UPDATE Cargo SET Name = :new_name "
            "WHERE UPPER(Name) = :old_name"
        ),
        {"new_name": "Usu\u00e1rio", "old_name": "FUNCIONARIO"},
    )


def _seed_default_roles(connection: Any) -> None:
    """Garante os cargos usados pelas permissÃµes padrÃ£o da aplicaÃ§Ã£o."""
    table = Cargo.__table__
    roles = (
        (1, "Admin"),
        (2, "Gerente"),
        (3, "Usu\u00e1rio"),
    )
    for role_id, name in roles:
        role = connection.execute(
            select(table.c["ID"]).where(table.c["ID"] == role_id)
        ).first()
        if role is None:
            connection.execute(
                insert(table).values(
                    ID=role_id,
                    Name=name,
                    Is_Active=True,
                )
            )
        else:
            connection.execute(
                update(table)
                .where(table.c["ID"] == role_id)
                .values(Name=name, Is_Active=True)
            )


def _migrate_legacy_passwords(connection: Any) -> None:
    """Converte senhas legadas em texto puro para hashes com salt."""
    table = Usuario.__table__
    password_column = table.c["Password"]
    rows = connection.execute(select(table.c["ID"], password_column)).all()
    for user_id, password in rows:
        if password and not password.startswith(("pbkdf2:", "scrypt:")):
            connection.execute(
                update(table)
                .where(table.c["ID"] == user_id)
                .values(Password=generate_password_hash(password, method="scrypt"))
            )


def _seed_initial_admin(connection: Any) -> None:
    """Cria o primeiro administrador somente com credenciais fornecidas no ambiente."""
    username = os.getenv("INITIAL_ADMIN_USERNAME")
    password = os.getenv("INITIAL_ADMIN_PASSWORD")
    name = os.getenv("INITIAL_ADMIN_NAME")
    if not username or not password or not name:
        return
    table = Usuario.__table__
    exists = connection.execute(
        select(table.c["ID"]).where(table.c["Cargo_ID"] == 1)
    ).first()
    if exists:
        return
    connection.execute(
        insert(table).values(
            Name=name.strip(),
            Username=username.strip(),
            Password=generate_password_hash(password, method="scrypt"),
            Is_Active=True,
            Cargo_ID=1,
        )
    )


def _seed_default_configuration(connection: Any) -> None:
    table = ConfiguracaoRegras.__table__
    if connection.execute(select(table.c.id_configuracao).limit(1)).first():
        return
    connection.execute(
        insert(table).values(
            id_configuracao=1,
            limiar_giro=10,
            dias_validade=7,
            margem_minima=15,
        )
    )


def migrate_database(app: Any) -> None:
    """Cria tabelas novas e atualiza colunas conforme o metadata do ORM."""
    _create_database_if_needed(app)

    # create_all cobre o primeiro bootstrap e deixa o autogerador focado em updates.
    db.create_all()

    with db.engine.begin() as connection:
        _ensure_user_auth_columns(connection)
        _ensure_stock_user_column(connection)
        context = MigrationContext.configure(
            connection,
            opts={
                "compare_type": True,
                "include_object": _include_object,
            },
        )
        migration = produce_migrations(context, db.metadata)
        if not migration.upgrade_ops.is_empty():
            existing_tables = {
                table_name.lower() for table_name in inspect(connection).get_table_names()
            }
            existing_indexes = _get_existing_indexes(connection)
            _apply_operations(
                Operations(context),
                migration.upgrade_ops.ops,
                existing_tables,
                existing_indexes,
            )
        _seed_default_roles(connection)
        _rename_roles(connection)
        _migrate_legacy_passwords(connection)
        _seed_initial_admin(connection)
        _seed_default_configuration(connection)

from datetime import timezone

from sqlalchemy import inspect, select

from app import db
from app.models.estoque import Estoque
from app.models.auditoria import Auditoria
from app.models.feedback_cliente import FeedbackCliente
from app.models.historico_preco import HistoricoPreco
from app.models.produto import Produto
from app.models.usuario import Usuario


class HistoricoRepository:
    """Combina registros persistidos em uma linha do tempo de atividades."""

    @staticmethod
    def _data_iso(data):
        if data.tzinfo is None:
            data = data.replace(tzinfo=timezone.utc)
        return data.isoformat()

    def listar(self):
        eventos = []
        usuarios = {
            usuario.id: usuario.name
            for usuario in db.session.execute(select(Usuario)).scalars()
        }
        produtos = {
            str(produto.id_produto): produto.nome
            for produto in db.session.execute(select(Produto)).scalars()
        }
        atores_feedback = {
            auditoria.entidade_id: auditoria.usuario_id
            for auditoria in db.session.execute(
                select(Auditoria).where(Auditoria.acao == "FEEDBACK_REGISTRADO")
            ).scalars()
        }

        inspector = inspect(db.engine)
        nome_tabela_estoque = next(
            (
                name
                for name in inspector.get_table_names()
                if name.lower() == Estoque.__tablename__.lower()
            ),
            Estoque.__tablename__,
        )
        estoque_columns = {
            column["name"].lower()
            for column in inspector.get_columns(nome_tabela_estoque)
        }
        colunas_movimentacao = [
            Estoque.id_estoque,
            Estoque.id_produto,
            Estoque.tipo_operacao,
            Estoque.quantidade,
            Estoque.motivo,
            Estoque.data_operacao,
        ]
        possui_responsavel_estoque = "usuario_id" in estoque_columns
        if possui_responsavel_estoque:
            colunas_movimentacao.append(Estoque.usuario_id)

        movimentacoes = db.session.execute(
            select(*colunas_movimentacao, Produto.nome).join(
                Produto, Estoque.id_produto == Produto.id_produto
            )
        ).all()
        for linha in movimentacoes:
            id_estoque, id_produto, operacao, quantidade, motivo, data_operacao = linha[:6]
            usuario_id = linha[6] if possui_responsavel_estoque else None
            nome_produto = linha[-1]
            eventos.append(
                {
                    "tipo": "ESTOQUE",
                    "data": self._data_iso(data_operacao),
                    "id_produto": id_produto,
                    "produto": nome_produto,
                    "tipo_operacao": operacao,
                    "quantidade": float(quantidade),
                    "motivo": motivo,
                    "usuario_id": usuario_id,
                    "usuario_nome": usuarios.get(usuario_id),
                    "responsavel_registrado": possui_responsavel_estoque,
                }
            )

        feedbacks = db.session.execute(
            select(FeedbackCliente, Produto.nome)
            .join(Produto, FeedbackCliente.id_produto == Produto.id_produto)
        ).all()
        for feedback, nome_produto in feedbacks:
            eventos.append(
                {
                    "tipo": "FEEDBACK",
                    "data": self._data_iso(feedback.data_criacao),
                    "id_produto": feedback.id_produto,
                    "produto": nome_produto,
                    "cliente": feedback.cliente,
                    "mensagem": feedback.comentario,
                    "usuario_id": atores_feedback.get(str(feedback.id_feedback)),
                    "usuario_nome": usuarios.get(
                        atores_feedback.get(str(feedback.id_feedback))
                    ),
                }
            )

        alteracoes_preco = db.session.execute(
            select(HistoricoPreco, Produto.nome)
            .join(Produto, HistoricoPreco.id_produto == Produto.id_produto)
        ).all()
        for alteracao, nome_produto in alteracoes_preco:
            eventos.append(
                {
                    "tipo": "PRECO",
                    "data": self._data_iso(alteracao.data_alteracao),
                    "id_produto": alteracao.id_produto,
                    "produto": nome_produto,
                    "preco_antigo": float(alteracao.preco_antigo),
                    "preco_novo": float(alteracao.preco_novo),
                    "motivo": alteracao.motivo,
                    "usuario_id": alteracao.usuario_id,
                    "usuario_nome": usuarios.get(alteracao.usuario_id),
                }
            )

        eventos_representados = {
            "ESTOQUE_MOVIMENTADO",
            "VENDA_REGISTRADA",
            "FEEDBACK_REGISTRADO",
        }
        auditorias = db.session.execute(
            select(Auditoria).order_by(Auditoria.data_criacao.desc())
        ).scalars()
        for auditoria in auditorias:
            if auditoria.acao in eventos_representados:
                continue
            detalhes = auditoria.detalhes or {}
            entidade = auditoria.entidade
            if entidade == "Produto":
                entidade = produtos.get(auditoria.entidade_id, entidade)
            elif entidade in {"SugestaoPreco", "PrecoConcorrente"}:
                id_produto = detalhes.get("id_produto")
                entidade = produtos.get(
                    str(id_produto),
                    "SugestÃ£o de preÃ§o"
                    if entidade == "SugestaoPreco"
                    else "PreÃ§o de concorrente",
                )
            elif entidade == "Usuario":
                entidade = (
                    detalhes.get("username")
                    or usuarios.get(auditoria.usuario_id)
                    or entidade
                )
            eventos.append(
                {
                    "tipo": "ATIVIDADE",
                    "data": self._data_iso(auditoria.data_criacao),
                    "id_produto": None,
                    "produto": entidade,
                    "acao": auditoria.acao,
                    "usuario_id": auditoria.usuario_id,
                    "usuario_nome": usuarios.get(auditoria.usuario_id),
                    "mensagem": detalhes,
                }
            )

        return sorted(eventos, key=lambda evento: evento["data"], reverse=True)

import re
import unicodedata
from decimal import Decimal, InvalidOperation
from pathlib import Path

from openpyxl import load_workbook
from sqlalchemy import select

from app import db
from app.config import Config
from app.models.preco_concorrente_planilha import PrecoConcorrentePlanilha


class PlanilhaConcorrenteRepository:
    COLUNAS_PRODUTO = {
        "produto",
        "nomeproduto",
        "nomedoproduto",
        "nome",
        "descricao",
        "item",
        "name",
        "product",
        "productname",
    }
    COLUNAS_VALOR = {
        "valor",
        "preco",
        "valorproduto",
        "valordoproduto",
        "value",
        "precoproduto",
        "precodoproduto",
        "price",
        "unitprice",
        "valorunitario",
    }

    def __init__(self, pasta=None):
        self.pasta = Path(pasta or Config.PLANILHAS_CONCORRENTES_DIR)

    @staticmethod
    def _normalizar_cabecalho(valor):
        texto = unicodedata.normalize("NFKD", str(valor or ""))
        texto = "".join(
            caractere for caractere in texto if not unicodedata.combining(caractere)
        )
        return re.sub(r"[^a-z0-9]", "", texto.lower())

    @staticmethod
    def _converter_valor(valor, arquivo, aba, linha):
        if isinstance(valor, bool) or valor is None:
            raise ValueError(
                f"Valor invÃ¡lido em {arquivo}, aba '{aba}', linha {linha}."
            )
        if isinstance(valor, (int, float, Decimal)):
            try:
                convertido = Decimal(str(valor))
            except InvalidOperation as error:
                raise ValueError(
                    f"Valor invÃ¡lido em {arquivo}, aba '{aba}', linha {linha}."
                ) from error
        else:
            texto = str(valor).strip().replace("R$", "").replace(" ", "")
            if "," in texto:
                texto = texto.replace(".", "").replace(",", ".")
            try:
                convertido = Decimal(texto)
            except InvalidOperation as error:
                raise ValueError(
                    f"Valor invÃ¡lido em {arquivo}, aba '{aba}', linha {linha}."
                ) from error
        if not convertido.is_finite() or convertido < 0:
            raise ValueError(
                f"Valor invÃ¡lido em {arquivo}, aba '{aba}', linha {linha}."
            )
        return convertido.quantize(Decimal("0.01"))

    def _ler_arquivo(self, caminho):
        workbook = load_workbook(caminho, data_only=True, read_only=True)
        registros = []
        try:
            for sheet in workbook.worksheets:
                linhas = enumerate(sheet.iter_rows(values_only=True), start=1)
                cabecalho = None
                coluna_produto = None
                coluna_valor = None
                linha_cabecalho = None

                for numero_linha, valores in linhas:
                    colunas = {
                        self._normalizar_cabecalho(valor): indice
                        for indice, valor in enumerate(valores)
                    }
                    coluna_produto = next(
                        (
                            indice
                            for nome, indice in colunas.items()
                            if nome in self.COLUNAS_PRODUTO
                        ),
                        None,
                    )
                    coluna_valor = next(
                        (
                            indice
                            for nome, indice in colunas.items()
                            if nome in self.COLUNAS_VALOR
                        ),
                        None,
                    )
                    if coluna_produto is not None and coluna_valor is not None:
                        cabecalho = valores
                        linha_cabecalho = numero_linha
                        break

                if cabecalho is None:
                    if sheet.max_row == 1 and sheet.max_column == 1:
                        continue
                    raise ValueError(
                        f"A planilha '{caminho.name}', aba '{sheet.title}', "
                        "precisa ter uma linha de cabeÃ§alho para produto e valor "
                        "(por exemplo: Nome e Valor)."
                    )

                if linha_cabecalho is None:
                    continue

                for numero_linha, linha in enumerate(
                    (valores for _, valores in linhas),
                    start=linha_cabecalho + 1,
                ):
                    if max(coluna_produto, coluna_valor) >= len(linha):
                        continue
                    nome_produto = linha[coluna_produto]
                    valor = linha[coluna_valor]
                    if nome_produto is None and valor is None:
                        continue
                    if nome_produto is None or not str(nome_produto).strip():
                        raise ValueError(
                            f"Produto ausente em {caminho.name}, aba "
                            f"'{sheet.title}', linha {numero_linha}."
                        )

                    registros.append(
                        {
                            "nome_arquivo": caminho.name,
                            "nome_aba": sheet.title,
                            "linha_excel": numero_linha,
                            "nome_produto": str(nome_produto).strip(),
                            "valor": self._converter_valor(
                                valor, caminho.name, sheet.title, numero_linha
                            ),
                        }
                    )
        finally:
            workbook.close()
        return registros

    def sincronizar(self):
        self.pasta.mkdir(parents=True, exist_ok=True)
        arquivos = sorted(
            caminho
            for caminho in self.pasta.iterdir()
            if caminho.is_file()
            and caminho.suffix.lower() in {".xlsx", ".xlsm"}
            and not caminho.name.startswith("~$")
        )
        registros = [
            registro
            for arquivo in arquivos
            for registro in self._ler_arquivo(arquivo)
        ]

        arquivos_lidos = {arquivo.name for arquivo in arquivos}
        registros_existentes = db.session.scalars(
            select(PrecoConcorrentePlanilha)
        ).all()
        existentes_por_origem = {
            (registro.nome_arquivo, registro.nome_aba, registro.linha_excel): registro
            for registro in registros_existentes
        }

        origens_lidas = set()
        for chave, registro in existentes_por_origem.items():
            if registro.nome_arquivo not in arquivos_lidos:
                db.session.delete(registro)
                continue

        for dados in registros:
            chave = (
                dados["nome_arquivo"],
                dados["nome_aba"],
                dados["linha_excel"],
            )
            origens_lidas.add(chave)
            registro = existentes_por_origem.get(chave)
            if registro is None:
                db.session.add(PrecoConcorrentePlanilha(**dados))
            else:
                registro.nome_produto = dados["nome_produto"]
                registro.valor = dados["valor"]

        for chave, registro in existentes_por_origem.items():
            if chave not in origens_lidas:
                db.session.delete(registro)

        db.session.commit()
        return len(arquivos), len(registros)

    def listar_comparacao(self):
        registros = db.session.scalars(
            select(PrecoConcorrentePlanilha).order_by(
                PrecoConcorrentePlanilha.nome_arquivo,
                PrecoConcorrentePlanilha.nome_produto,
                PrecoConcorrentePlanilha.linha_excel,
            )
        ).all()

        arquivos = sorted({registro.nome_arquivo for registro in registros})
        produtos = {}
        chaves_produtos = {}
        for registro in registros:
            normalizado = self._normalizar_cabecalho(registro.nome_produto)
            if normalizado not in produtos:
                produtos[normalizado] = registro.nome_produto
                chaves_produtos[normalizado] = {}
            chaves_produtos[normalizado].setdefault(
                registro.nome_arquivo, []
            ).append(float(registro.valor))

        comparacoes = []
        for chave, nome_produto in produtos.items():
            precos = chaves_produtos[chave]
            valores = [valor for lista in precos.values() for valor in lista]
            comparacoes.append(
                {
                    "nome_produto": nome_produto,
                    "precos": {
                        arquivo: precos.get(arquivo, []) for arquivo in arquivos
                    },
                    "diferenca": (
                        round(max(valores) - min(valores), 2)
                        if len(valores) >= 2
                        else None
                    ),
                }
            )

        comparacoes.sort(key=lambda produto: produto["nome_produto"].casefold())
        return {"arquivos": arquivos, "produtos": comparacoes}

from app import db

class VendaRepository(BaseRepository[Venda]):
    def __init__(self):
        super().__init__(Venda)

    def total_vendido(self, id_produto: int, dias: int = 30) -> float:
        agora = datetime.utcnow()
        inicio = agora - timedelta(days=dias)
        total = (
            db.session.query(func.coalesce(func.sum(Venda.quantidade), 0))
            .filter(
                Venda.id_produto == id_produto,
                Venda.data_venda >= inicio,
                Venda.data_venda <= agora,
            )
            .scalar()
        )
        return float(total)

    def totais_por_produto(
            self, id_produtos: list[int], agora: datetime | None = None
    ) -> dict[int, tuple[float, float]]:
        if not id_produtos:
            return {}

        agora = agora or datetime.utcnow()
        inicio_semanal = agora - timedelta(days=7)
        inicio_mensal = agora - timedelta(days=30)
        semana = func.coalesce(
            func.sum(
                case(
                    (Venda.data_venda >= inicio_semanal, Venda.quantidade),
                    else_=0,
                )
            ),
            0,
        )
        mes = func.coalesce(func.sum(Venda.quantidade), 0)
        resultados = (
            db.session.query(Venda.id_produto, semana, mes)
            .filter(
                Venda.id_produto.in_(id_produtos),
                Venda.data_venda >= inicio_mensal,
                Venda.data_venda <= agora,
            )
            .group_by(Venda.id_produto)
            .all()
        )
        return {
            id_produto: (float(total_semanal), float(total_mensal))
            for id_produto, total_semanal, total_mensal in resultados
        }

            return None

            raise ValueError("Os dados do produto devem ser um objeto JSON.")

        if not nome:
            raise ValueError("O nome do produto Ã© obrigatÃ³rio.")
        codigo = dados.get("codigo", produto.codigo)
        codigo = str(codigo).strip().upper() or None if codigo is not None else None
        self._validar_nao_duplicado(nome, codigo, ignorar_id=id_produto)
        estoque_anterior = Decimal(str(produto.estoque))
        preco_anterior = Decimal(str(produto.preco_atual))
        produto.nome = nome
        for campo in ("custo", "preco_atual", "estoque"):
            if campo in dados:
                try:
                    valor = Decimal(str(dados[campo]))
                except (ArithmeticError, ValueError):
                    raise ValueError(f"{campo} deve ser numÃ©rico.") from None
                if not valor.is_finite():
                    raise ValueError(f"{campo} deve ser finito.")

                    ),

            )

function tokenExpirado(token) {
    try {
        const payload = JSON.parse(atob(token.split('.')[1].replace(/-/g, '+').replace(/_/g, '/')));
        return payload.exp ? payload.exp * 1000 <= Date.now() : true;
    } catch(e) {
        return true;
    }
}

function mostrarAplicacao() {
    document.getElementById('loginView').style.display = 'none';
    document.getElementById('app').classList.add('shown');
    document.getElementById('userName').textContent = user.nome || user.Name || 'Utilizador';
    document.getElementById('avatarLetter').textContent = (user.cargo_nome || user.Cargo_Name || '?')[0].toUpperCase();
}

async function iniciarSessao() {
    if (!accessToken || !user) return;
    mostrarAplicacao();
    if (typeof buildNav === 'function') buildNav();
    await refreshProdutos();
    if (typeof render === 'function') render();
}

async function restaurarSessao() {
    try {
        const sessao = JSON.parse(localStorage.getItem(SESSION_KEY));
        if (!sessao || !sessao.accessToken || !sessao.user || tokenExpirado(sessao.accessToken)) {
            localStorage.removeItem(SESSION_KEY);
            return;
        }
        accessToken = sessao.accessToken;
        user = sessao.user;
        await iniciarSessao();
    } catch(e) {
        localStorage.removeItem(SESSION_KEY);
    }
}

async function api(path, opts={}) {
    const headers = Object.assign({'Content-Type':'application/json'}, opts.headers||{});
    if (accessToken) headers.Authorization = 'Bearer ' + accessToken;
    const res = await fetch(API_BASE + path, Object.assign({}, opts, {headers}));
    let body = null;
    try { body = await res.json(); } catch(e){}
    if (!res.ok) {
        const error = new Error((body && (body.erro || body.message)) || ('Erro HTTP ' + res.status));
        error.campos = body && (body.campos || (body.campo ? [body.campo] : []));
        throw error;
    }
    return body;
}

async function doLogin(){
    const email = document.getElementById('loginEmail').value.trim();
    const senha = document.getElementById('loginSenha').value.trim();
    const usuarioInput = document.getElementById('loginEmail');
    const senhaInput = document.getElementById('loginSenha');
    const loginErr = document.getElementById('loginErr');
    const usuarioErr = document.getElementById('loginUsuarioErr');
    const senhaErr = document.getElementById('loginSenhaErr');
    usuarioInput.classList.remove('input-error');
    senhaInput.classList.remove('input-error');
    usuarioInput.setAttribute('aria-invalid', 'false');
    senhaInput.setAttribute('aria-invalid', 'false');
    usuarioErr.textContent = '';
    senhaErr.textContent = '';
    loginErr.textContent = '';
    if (!email || !senha) {
        if (!email) {
            usuarioInput.classList.add('input-error');
            usuarioInput.setAttribute('aria-invalid', 'true');
            usuarioErr.textContent = 'Informe o usuÃ¡rio.';
        }
        if (!senha) {
            senhaInput.classList.add('input-error');
            senhaInput.setAttribute('aria-invalid', 'true');
            senhaErr.textContent = 'Informe a senha.';
        }
        return;
    }

    try {
        const resp = await api('/auth/login', {
            method: 'POST',
            body: JSON.stringify({ Username: email, Password: senha })
        });
        accessToken = resp.access_token;
        user = resp.usuario;
        localStorage.setItem(SESSION_KEY, JSON.stringify({ accessToken, user }));
        await iniciarSessao();
    } catch(e) {
        const campos = e.campos || [];
        if (campos.includes('usuario')) {
            usuarioInput.classList.add('input-error');
            usuarioInput.setAttribute('aria-invalid', 'true');
            usuarioErr.textContent = 'UsuÃ¡rio incorreto.';
        }
        if (campos.includes('senha')) {
            senhaInput.classList.add('input-error');
            senhaInput.setAttribute('aria-invalid', 'true');
            senhaErr.textContent = 'Senha incorreta.';
        }
        if (campos.length === 0) loginErr.textContent = e.message;
    }
}

function logout(){

    document.getElementById('loginView').style.display = 'flex';

    if (!dateStr) return { label: 'VÃ¡lido', cls: 'valido' };
    const diff = (new Date(dateStr) - new Date()) / 86400000;
    if (diff < 0) return { label: 'Vencido', cls: 'vencido' };
    if (diff <= 7) return { label: 'Alerta', cls: 'alerta' };
    return { label: 'VÃ¡lido', cls: 'valido' };
}
function margem(p) {
    return p.custo > 0 ? ((p.preco_atual - p.custo) / p.custo) * 100 : 0;
}

const SIDEBAR_STATE_KEY = 'sgb_varejo_sidebar_collapsed';
const DISPLAY_MODE_KEY = 'sgb_varejo_display_mode';

function getDisplayMode() {
  return localStorage.getItem(DISPLAY_MODE_KEY) === 'list' ? 'list' : 'grid';
}

function collectionClass() {
  return `view-collection is-${getDisplayMode()}`;
}

function displayModeControl() {
  const mode = getDisplayMode();
  return `<div class="display-mode-control" role="group" aria-label="Modo de exibiÃ§Ã£o">
    <span>Exibir:</span>
    <button type="button" data-display-mode="grid" class="btn btn-outline${mode === 'grid' ? ' selected' : ''}" aria-pressed="${mode === 'grid'}" onclick="setDisplayMode('grid')">Grade</button>
    <button type="button" data-display-mode="list" class="btn btn-outline${mode === 'list' ? ' selected' : ''}" aria-pressed="${mode === 'list'}" onclick="setDisplayMode('list')">Lista</button>
  </div>`;
}

function setDisplayMode(mode) {
  if (mode !== 'grid' && mode !== 'list') return;
  localStorage.setItem(DISPLAY_MODE_KEY, mode);
  document.querySelectorAll('.view-collection').forEach((collection) => {
    collection.classList.toggle('is-grid', mode === 'grid');
    collection.classList.toggle('is-list', mode === 'list');
  });
  document.querySelectorAll('.display-mode-control button').forEach((button) => {
    const selected = button.dataset.displayMode === mode;
    button.classList.toggle('selected', selected);
    button.setAttribute('aria-pressed', String(selected));
  });
}

function atualizarBotaoSidebar() {
  const app = document.getElementById('app');
  const button = document.getElementById('sidebarToggle');
  if (!app || !button) return;
  const recolhida = app.classList.contains('sidebar-collapsed');
  button.textContent = recolhida ? 'â˜°' : 'Ã—';
  button.setAttribute('aria-label', recolhida ? 'Expandir menu' : 'Recolher menu');
  button.title = recolhida ? 'Expandir menu' : 'Recolher menu';

function alternarSidebar() {
  const app = document.getElementById('app');
  if (!app) return;
  const recolhida = app.classList.toggle('sidebar-collapsed');
  localStorage.setItem(SIDEBAR_STATE_KEY, recolhida ? 'true' : 'false');

}

function restaurarSidebar() {
  if (localStorage.getItem(SIDEBAR_STATE_KEY) === 'true') {
    document.getElementById('app').classList.add('sidebar-collapsed');
  }
  atualizarBotaoSidebar();
}

function escaparHtml(valor) {
  return String(valor).replace(/[&<>"']/g, (caractere) => ({

    '"': '&quot;',
    "'": '&#39;'
  })[caractere]);
}

function definirPeriodoRelatorio() {
  const hoje = new Date();
  const inicioMes = new Date(hoje.getFullYear(), hoje.getMonth(), 1);
  const iso = (data) => {
    const deslocada = new Date(data.getTime() - data.getTimezoneOffset() * 60000);
    return deslocada.toISOString().slice(0, 10);
  };
  const inicio = document.getElementById('relatorioInicio');
  if (!inicio) return;
  inicio.value = iso(inicioMes);
  document.getElementById('relatorioFim').value = iso(hoje);
}

import gzip
import os
import shutil
import subprocess
import sys
from datetime import datetime, timedelta, timezone
from pathlib import Path


def backup_database():
    database = os.getenv("DB_NAME", "sgb_varejo")
    host = os.getenv("DB_HOST", "localhost")
    port = os.getenv("DB_PORT", "3306")
    user = os.getenv("DB_USER", "root")
    password = os.getenv("DB_PASSWORD", "")
    executable = os.getenv("MYSQLDUMP_BIN", "mysqldump")
    backup_dir = Path(os.getenv("DB_BACKUP_DIR", "backups")).resolve()
    backup_dir.mkdir(parents=True, exist_ok=True)

    timestamp = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ")
    destination = backup_dir / f"sgpv-{timestamp}.sql.gz"
    temporary = destination.with_suffix(destination.suffix + ".tmp")
    command = [
        executable,
        "--single-transaction",
        "--routines",
        "--triggers",
        "--host",
        host,
        "--port",
        port,
        "--user",
        user,
        "--databases",
        database,
    ]
    environment = os.environ.copy()
    environment["MYSQL_PWD"] = password

    try:
        with temporary.open("wb") as raw_file:
            with gzip.GzipFile(fileobj=raw_file, mode="wb") as compressed:
                result = subprocess.run(
                    command,
                    stdout=compressed,
                    stderr=subprocess.PIPE,
                    env=environment,
                    check=False,
                    timeout=7200,
                )
        if result.returncode != 0:
            message = result.stderr.decode("utf-8", errors="replace").strip()
            raise RuntimeError(f"mysqldump falhou (cÃ³digo {result.returncode}): {message}")
        temporary.replace(destination)
    except Exception:
        temporary.unlink(missing_ok=True)
        raise

    limite = datetime.now(timezone.utc) - timedelta(days=30)
    for arquivo in backup_dir.glob("sgpv-*.sql.gz"):
        modificado = datetime.fromtimestamp(arquivo.stat().st_mtime, timezone.utc)
        if modificado < limite:
            arquivo.unlink()
    return destination


if __name__ == "__main__":
    try:
        print(backup_database())
    except (OSError, RuntimeError, subprocess.TimeoutExpired) as error:
        print(f"Falha no backup do banco: {error}", file=sys.stderr)
        raise SystemExit(1) from error

import gzip

from scripts.backup_mysql import backup_database


class BackupTests(unittest.TestCase):
    def test_creates_compressed_dump_and_removes_expired_backups(self):
        with tempfile.TemporaryDirectory() as directory:
            old_backup = Path(directory) / "sgpv-old.sql.gz"
            old_backup.touch()
            old_time = 1
            os.utime(old_backup, (old_time, old_time))

            def mock_dump(*args, stdout, **kwargs):
                stdout.write(b"sql dump")
                return type("Result", (), {"returncode": 0, "stderr": b""})()

            with (
                patch.dict(os.environ, {"DB_BACKUP_DIR": directory}),
                patch("scripts.backup_mysql.subprocess.run", side_effect=mock_dump),
            ):
                created = backup_database()

            self.assertTrue(created.exists())
            with gzip.open(created, "rb") as backup_file:
                self.assertEqual(backup_file.read(), b"sql dump")
            self.assertFalse(old_backup.exists())

    def test_history_includes_timestamp_actor_and_stock_activity(self):

            },
            headers=self._headers(self.manager),
        )
        self.assertEqual(response.status_code, 201, response.get_json())

        inspector_real = inspect(db.engine)
        with patch("app.repositories.historico_repository.inspect") as inspector_mock:
            inspector_mock.return_value.get_table_names.return_value = (
                inspector_real.get_table_names()
            )
            inspector_mock.return_value.get_columns.side_effect = lambda nome_tabela: [
                coluna
                for coluna in inspector_real.get_columns(nome_tabela)
                if coluna["name"].lower() != "usuario_id"
            ]
            history_legacy = self.client.get(
                "/api/historico", headers=self._headers(self.manager)
            )

        self.assertEqual(history_legacy.status_code, 200, history_legacy.get_json())

            "/api/historico", headers=self._headers(self.manager)
        )

        self.assertEqual(history.status_code, 200, history.get_json())
        event = next(
            item
            for item in history.get_json()
            if item["tipo"] == "ESTOQUE"
            and item["motivo"] == "Uso de teste"
        )
        self.assertTrue(event["data"])
        self.assertEqual(event["usuario_nome"], "Gerente")
        self.assertEqual(event["produto"], product["nome"])

    def test_legacy_stock_table_migration_adds_user_column(self):
